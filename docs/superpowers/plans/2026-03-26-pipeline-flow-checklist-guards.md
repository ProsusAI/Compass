# Pipeline Flow: Checklist Tool & Precondition Guards — Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a stateless pipeline checklist tool and precondition guards to every MCP tool so external LLM agents can't call tools out of order.

**Architecture:** A `pipeline_status` module derives pipeline progress from file existence checks under `outputs/<run_id>/`. A `pipeline_guards` module provides a decorator and helpers that reject tool calls when prerequisites are missing. All pipeline artifacts move to run-scoped named subfolders. The `run_id` is generated by `submit_input_report` and threaded through all tool calls.

**Tech Stack:** Python 3.11+, Pydantic, FastMCP, pytest, pytest-asyncio

**Spec:** `docs/superpowers/specs/2026-03-26-pipeline-flow-checklist-guards-design.md`

# odysseus/agents/pipeline_guards.py
"""Precondition guards for MCP tools.

Each guard checks that prerequisite artifacts exist on disk before
allowing a tool to execute. On failure, raises ToolError with an
actionable message pointing to get_pipeline_status.
"""

from __future__ import annotations

import asyncio
import functools
from pathlib import Path
from typing import Any, Callable

from mcp.server.fastmcp.exceptions import ToolError


def _build_error(paths: list[Path | str], stage: int, stage_name: str, hint: str) -> ToolError:
    missing = [str(p) for p in paths if not Path(p).is_file()]
    if not missing:
        return None  # type: ignore[return-value]
    return ToolError(
        f"Pipeline precondition not met: missing {', '.join(missing)}. "
        f"You are at stage {stage} ({stage_name}). {hint} "
        f"Call get_pipeline_status for the full checklist."
    )


def require_artifacts(
    *paths: Path | str,
    stage: int,
    stage_name: str,
    hint: str,
) -> Callable:
    """Decorator that checks file existence before tool execution.

    Works with both sync and async functions (required for MCP tools).

    Args:
        *paths: Artifact paths that must exist.
        stage: Pipeline stage number for the error message.
        stage_name: Human-readable stage name.
        hint: What the agent should do instead.
    """

    def decorator(fn: Callable) -> Callable:
        if asyncio.iscoroutinefunction(fn):
            @functools.wraps(fn)
            async def async_wrapper(*args: Any, **kwargs: Any) -> Any:
                err = _build_error(list(paths), stage, stage_name, hint)
                if err is not None:
                    raise err
                return await fn(*args, **kwargs)
            return async_wrapper
        else:
            @functools.wraps(fn)
            def sync_wrapper(*args: Any, **kwargs: Any) -> Any:
                err = _build_error(list(paths), stage, stage_name, hint)
                if err is not None:
                    raise err
                return fn(*args, **kwargs)
            return sync_wrapper

    return decorator


def check_artifacts(
    *paths: Path | str,
    stage: int,
    stage_name: str,
    hint: str,
) -> None:
    """Inline guard — call at the top of a tool function.

    Same logic as the decorator but for tools that need dynamic paths
    (computed from parameters).

    Raises:
        ToolError: If any path does not exist.
    """
    missing = [str(p) for p in paths if not Path(p).is_file()]
    if missing:
        raise ToolError(
            f"Pipeline precondition not met: missing {', '.join(missing)}. "
            f"You are at stage {stage} ({stage_name}). {hint} "
            f"Call get_pipeline_status for the full checklist."
        )
```

# odysseus/agents/pipeline_status.py
"""Stateless pipeline progress detection.

Derives pipeline stage completion from artifact existence on disk.
No state files — the filesystem is the source of truth.
"""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any

from odysseus.project_dir import get_project_dir

# Stage definitions: (stage_number, name, subfolder, required_files)
_STAGES: list[tuple[int, str, str, list[str]]] = [
    (1, "Input Report", "input", ["input_report.md"]),
    (2, "Data Validated", "validation", [
        "transformed.jsonl", "data_quality_report.json", "routing_context.json",
    ]),
    (3, "Routing Analysis & Split", "analysis", [
        "validation_report.json", "dev.jsonl", "holdout.jsonl",
        "dev_rationale_card_set.json", "holdout_rationale_card_set.json",
        "vocabulary_registry.json",
    ]),
    (4, "Backend Configured", "", []),  # special: checks backends/*.yaml
    (5, "Prompt v1 Compiled", "prompts", []),  # special: globs v1.*
    (6, "Eval Loop Active", "search", ["search_state.json"]),  # + round check
    (7, "Converged", "search", ["search_state.json"]),  # + converged check
    (8, "Holdout Validation", "", []),  # future
    (9, "Final Report", "", []),  # future
]


def discover_runs(outputs_dir: Path) -> list[str]:
    """Return run IDs sorted by most recent first.

    A run is detected by the presence of <run_id>/input/input_report.md.
    """
    if not outputs_dir.is_dir():
        return []
    runs: list[tuple[float, str]] = []
    for child in outputs_dir.iterdir():
        if not child.is_dir():
            continue
        report = child / "input" / "input_report.md"
        if report.is_file():
            runs.append((report.stat().st_mtime, child.name))
    runs.sort(reverse=True)
    return [name for _, name in runs]


def _check_stage(
    run_dir: Path,
    project_dir: Path,
    stage_num: int,
    name: str,
    subfolder: str,
    required_files: list[str],
) -> dict[str, Any]:
    """Check completion of a single stage."""
    # Special cases
    if stage_num == 4:
        backends_dir = project_dir / "backends"
        yamls = list(backends_dir.glob("*.yaml")) if backends_dir.is_dir() else []
        return {
            "stage": stage_num,
            "name": name,
            "status": "complete" if yamls else "incomplete",
            "artifacts": [str(p) for p in yamls],
        }
    if stage_num == 5:
        prompts_dir = run_dir / "prompts"
        matches = list(prompts_dir.glob("v1.*")) if prompts_dir.is_dir() else []
        return {
            "stage": stage_num,
            "name": name,
            "status": "complete" if matches else "incomplete",
            "artifacts": [str(p) for p in matches],
        }
    if stage_num in (6, 7):
        ss_path = run_dir / "search" / "search_state.json"
        if not ss_path.is_file():
            return {"stage": stage_num, "name": name, "status": "incomplete", "artifacts": []}
        state = json.loads(ss_path.read_text())
        if stage_num == 6:
            ok = state.get("round", 0) >= 1
        else:
            ok = state.get("converged", False)
        return {
            "stage": stage_num,
            "name": name,
            "status": "complete" if ok else "incomplete",
            "artifacts": [str(ss_path)] if ok else [],
        }
    if stage_num in (8, 9):
        return {"stage": stage_num, "name": name, "status": "incomplete", "artifacts": []}

    # Standard: check all required files in subfolder
    stage_dir = run_dir / subfolder
    found = []
    missing = []
    for f in required_files:
        p = stage_dir / f
        if p.is_file():
            found.append(str(p))
        else:
            missing.append(f)

    if not missing:
        return {"stage": stage_num, "name": name, "status": "complete", "artifacts": found}

    # Stage 3 in-progress detection: check scratch checkpoints
    if stage_num == 3:
        scratch_dir = project_dir / "scratch"
        if scratch_dir.is_dir():
            for phase_file in ["phase3_validated.json", "phase2_rationale.json", "phase1_classification.json"]:
                checkpoints = list(scratch_dir.glob(f"*/{phase_file}"))
                if checkpoints:
                    return {
                        "stage": stage_num, "name": name, "status": "in_progress",
                        "artifacts": found,
                        "detail": f"Checkpoint found: {checkpoints[0]}",
                    }

    return {"stage": stage_num, "name": name, "status": "incomplete", "artifacts": found}


def get_pipeline_status(
    outputs_dir: Path | None = None,
    run_id: str | None = None,
) -> dict[str, Any]:
    """Derive pipeline status from artifacts on disk.

    Args:
        outputs_dir: Root outputs directory. Defaults to project_dir / "outputs".
        run_id: Specific run to check. If None, uses most recent.

    Returns:
        Dict with run_id, stages, current_stage, next_action, etc.
    """
    project_dir = get_project_dir()
    if outputs_dir is None:
        outputs_dir = project_dir / "outputs"

    # Resolve run_id
    if run_id is None:
        runs = discover_runs(outputs_dir)
        if not runs:
            return {
                "run_id": None,
                "stages": [],
                "current_stage": 1,
                "current_stage_name": "Input Report",
                "next_action": "No pipeline runs found. Call submit_input_report to start.",
                "available_tools": ["submit_input_report"],
                "available_prompts": ["odysseus_routing_input"],
            }
        run_id = runs[0]

    run_dir = outputs_dir / run_id

    # Check each stage
    stages = []
    current_stage = None
    for stage_num, name, subfolder, required_files in _STAGES:
        result = _check_stage(run_dir, project_dir, stage_num, name, subfolder, required_files)
        stages.append(result)
        if current_stage is None and result["status"] != "complete":
            current_stage = stage_num

    # Mark stages after current as blocked
    if current_stage is not None:
        for s in stages:
            if s["stage"] > current_stage and s["status"] == "incomplete":
                s["status"] = "blocked"

    if current_stage is None:
        current_stage = 9  # all done

    current_name = next((s["name"] for s in stages if s["stage"] == current_stage), "")
    next_action, tools, prompts = _next_action_for_stage(current_stage, run_id)

    return {
        "run_id": run_id,
        "stages": stages,
        "current_stage": current_stage,
        "current_stage_name": current_name,
        "next_action": next_action,
        "available_tools": tools,
        "available_prompts": prompts,
    }


def _next_action_for_stage(
    stage: int, run_id: str
) -> tuple[str, list[str], list[str]]:
    """Return (next_action, available_tools, available_prompts) for a stage."""
    mapping: dict[int, tuple[str, list[str], list[str]]] = {
        1: (
            "Activate odysseus_routing_input prompt to collect problem spec and dataset path.",
            ["submit_input_report"],
            ["odysseus_routing_input"],
        ),
        2: (
            f"Activate odysseus_data_validation prompt. run_id={run_id}",
            ["detect_and_parse_dataset", "transform_dataset", "validate_dataset"],
            ["odysseus_data_validation"],
        ),
        3: (
            f"Activate odysseus_routing_analysis prompt. run_id={run_id}",
            ["create_seed_registry_tool", "resolve_registry_tool",
             "validate_rationale_card_set_tool", "prune_registry_tool",
             "stratified_split_tool"],
            ["odysseus_routing_analysis"],
        ),
        4: (
            "Activate odysseus_backend_setup prompt to configure an LLM backend.",
            [],
            ["odysseus_backend_setup"],
        ),
        5: (
            f"Activate odysseus_prompt_builder prompt. run_id={run_id}",
            ["init_search_state_tool", "register_candidate_tool", "run_eval"],
            ["odysseus_prompt_builder"],
        ),
        6: (
            f"Continue the optimization loop. run_id={run_id}",
            ["register_candidate_tool", "run_eval", "advance_round_tool",
             "build_review_briefing_tool", "record_directive_outcomes_tool"],
            ["odysseus_prompt_builder", "odysseus_review_agent"],
        ),
        7: (
            f"Search converged. Run holdout validation. run_id={run_id}",
            ["filter_holdout_dataset_tool", "run_holdout_eval"],
            [],
        ),
    }
    return mapping.get(stage, ("Pipeline stage not yet implemented.", [], []))
```

```python
@mcp.tool()
async def submit_input_report(
    report: str,
    dataset_path: str,
    problem_description: str,
    bootstrap_from_run_id: str | None = None,
) -> str:
    """[Stage 1: Input] Submit a validated input report to the pipeline.

    No prerequisites. Generates a run_id that scopes all subsequent
    pipeline artifacts. Returns the run_id for use in all downstream
    tool calls.

    Args:
        report: The full validated input report (Markdown).
        dataset_path: Absolute filesystem path to the routing dataset.
        problem_description: The validated problem description.
        bootstrap_from_run_id: Optional run_id to copy seed prompt from.

    Returns:
        JSON with run_id and persisted report path.
    """
    # Note: add `import uuid` to the module-level imports in mcp.py

    if not report.strip():
        raise ToolError("submit_input_report failed: report is empty")
    if not dataset_path.strip():
        raise ToolError("submit_input_report failed: dataset_path is empty")
    if not problem_description.strip():
        raise ToolError("submit_input_report failed: problem_description is empty")

    run_id = uuid.uuid4().hex[:8]
    project_dir = get_project_dir()
    input_dir = project_dir / "outputs" / run_id / "input"
    input_dir.mkdir(parents=True, exist_ok=True)

    report_path = input_dir / "input_report.md"
    report_path.write_text(report, encoding="utf-8")

    # Bootstrap: copy best prompt from a prior run
    if bootstrap_from_run_id:
        source_prompts = project_dir / "outputs" / bootstrap_from_run_id / "prompts"
        if source_prompts.is_dir():
            dest_prompts = project_dir / "outputs" / run_id / "prompts"
            dest_prompts.mkdir(parents=True, exist_ok=True)
            # Find highest version prompt
            prompt_files = sorted(source_prompts.glob("v*.txt"))
            if prompt_files:
                latest = prompt_files[-1]
                (dest_prompts / "bootstrap.txt").write_text(
                    latest.read_text(encoding="utf-8"), encoding="utf-8"
                )

    return json.dumps({
        "run_id": run_id,
        "report_path": str(report_path),
        "dataset_path": dataset_path,
    })
```
