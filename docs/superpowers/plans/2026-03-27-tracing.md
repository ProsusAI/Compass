# Tracing Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add opt-in MCP tool-call tracing to the dev branch so developers can verify pipeline flow correctness via structured JSONL files and a CLI waterfall view.

**Architecture:** A `Tracer` class in `odysseus/tracing.py` writes append-only JSON events to `outputs/<run_id>/traces/events.jsonl`. A process-level singleton `_tracer` (constructed lazily from `get_project_dir()`) is shared across all tool calls; `run_id` is passed per-call so a single tracer instance handles multiple runs in one server process. A `@trace_tool` decorator (enabled by `ODYSSEUS_TRACE=1`) wraps each tool handler in `mcp.py`. A `trace` subcommand in `cli.py` formats the JSONL as a human-readable waterfall.

> **Design note:** The spec's class diagram shows `Tracer.__init__(run_id: str | None)` — this was a sketch. The final design takes `project_dir: Path` in `__init__` and `run_id` in each method call so the singleton can serve multiple runs without reinstantiation.

**Tech Stack:** Python 3.11+, `itertools.count`, `pathlib.Path`, stdlib `json`/`time`/`datetime`/`functools`. No new dependencies. `rich` used for CLI colour if available (already a transitive dep of `mcp[cli]`).

```python
"""Opt-in MCP tool-call tracing for the Odysseus dev branch.

Enabled by setting ODYSSEUS_TRACE=1 in the environment.
Events are written to outputs/<run_id>/traces/events.jsonl,
or outputs/traces/session-<timestamp>.jsonl when run_id is unknown.
"""

from __future__ import annotations

import functools
import itertools
import json
import os
import time
from collections.abc import Callable
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

_MAX_ARG_LEN = 500


def _now() -> str:
    return datetime.now(timezone.utc).isoformat()


def _sanitize_args(args: dict) -> dict:
    """Truncate string values longer than _MAX_ARG_LEN chars."""
    result = {}
    for k, v in args.items():
        if isinstance(v, str) and len(v) > _MAX_ARG_LEN:
            result[k] = v[:_MAX_ARG_LEN] + "…"
        else:
            result[k] = v
    return result


class Tracer:
    """Writes tool-call trace events to JSONL files under project_dir/outputs/.

    Takes project_dir at construction time; run_id is passed per-call so a
    single singleton instance can serve multiple pipeline runs.
    """

    def __init__(self, project_dir: Path) -> None:
        self._project_dir = project_dir
        self._counter = itertools.count(1)
        self._session_file: Path | None = None

    def tool_start(self, tool: str, run_id: str | None, args: dict) -> int:
        """Record a tool_start event. Returns the seq number for this call."""
        seq = next(self._counter)
        self._write(
            run_id,
            {
                "seq": seq,
                "timestamp": _now(),
                "event": "tool_start",
                "tool": tool,
                "run_id": run_id,
                "args": _sanitize_args(args),
            },
        )
        return seq

    def tool_end(
        self,
        seq: int,
        tool: str,
        run_id: str | None,
        result_summary: dict,
        duration_ms: int,
    ) -> None:
        """Record a tool_end event."""
        self._write(
            run_id,
            {
                "seq": seq,
                "timestamp": _now(),
                "event": "tool_end",
                "tool": tool,
                "run_id": run_id,
                "duration_ms": duration_ms,
                "status": "ok",
                "result_summary": result_summary,
            },
        )

    def tool_error(
        self,
        seq: int,
        tool: str,
        run_id: str | None,
        error: str,
        duration_ms: int,
    ) -> None:
        """Record a tool_error event."""
        self._write(
            run_id,
            {
                "seq": seq,
                "timestamp": _now(),
                "event": "tool_error",
                "tool": tool,
                "run_id": run_id,
                "duration_ms": duration_ms,
                "status": "error",
                "error": error,
            },
        )

    def _get_trace_file(self, run_id: str | None) -> Path:
        if run_id:
            path = self._project_dir / "outputs" / run_id / "traces" / "events.jsonl"
        else:
            if self._session_file is None:
                ts = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ")
                self._session_file = (
                    self._project_dir / "outputs" / "traces" / f"session-{ts}.jsonl"
                )
            path = self._session_file
        path.parent.mkdir(parents=True, exist_ok=True)
        return path

    def _write(self, run_id: str | None, event: dict) -> None:
        path = self._get_trace_file(run_id)
        with path.open("a", encoding="utf-8") as f:
            f.write(json.dumps(event) + "\n")
```

```python
# ---------------------------------------------------------------------------
# Process-level singleton
# ---------------------------------------------------------------------------

_tracer: "Tracer | None" = None


def _get_tracer() -> Tracer:
    """Return the process-level Tracer, constructing it on first use."""
    global _tracer
    if _tracer is None:
        from odysseus.project_dir import get_project_dir
        _tracer = Tracer(get_project_dir())
    return _tracer


# ---------------------------------------------------------------------------
# Result summary extractors
# Each callable receives (result_str: str, kwargs: dict) -> dict[str, Any].
# Must never raise: bad input returns {}.
# ---------------------------------------------------------------------------

def _json_keys(keys: list[str]) -> Callable[[str, dict], dict]:
    """Return an extractor that parses JSON and picks specific top-level keys."""
    def extract(result: str, kwargs: dict) -> dict:
        try:
            data = json.loads(result)
            return {k: data[k] for k in keys if k in data}
        except Exception:
            return {}
    return extract


def _extract_run_eval(result: str, kwargs: dict) -> dict:
    try:
        data = json.loads(result)
        metrics = data.get("metrics", {})
        summary = data.get("summary", {})
        return {
            "accuracy": metrics.get("accuracy"),
            "error_count": summary.get("error_count"),
            "total_examples": summary.get("total_examples"),
            "cost_usd": summary.get("cost_usd"),
        }
    except Exception:
        return {}


def _extract_resolve_registry(result: str, kwargs: dict) -> dict:
    try:
        data = json.loads(result)
        if not data.get("found", True):
            return {"found": False, "entry_count": 0}
        entry_count = (
            len(data.get("intent_pattern", []))
            + len(data.get("complexity_structure", []))
            + len(data.get("ambiguity_tags", []))
        )
        return {"found": True, "entry_count": entry_count}
    except Exception:
        return {}


def _extract_validate_dataset(result: str, kwargs: dict) -> dict:
    try:
        data = json.loads(result)
        findings = data.get("findings", [])
        errors = sum(1 for f in findings if f.get("severity") == "error")
        warnings = sum(1 for f in findings if f.get("severity") == "warning")
        return {"passed": errors == 0, "error_count": errors, "warning_count": warnings}
    except Exception:
        return {}


def _extract_validate_card_set(result: str, kwargs: dict) -> dict:
    try:
        data = json.loads(result)
        errors = data.get("error_count", 0)
        return {"passed": errors == 0, "error_count": errors}
    except Exception:
        return {}


def _extract_seed_registry(result: str, kwargs: dict) -> dict:
    try:
        data = json.loads(result)
        entry_count = (

            + len(data.get("complexity_structure", []))
            + len(data.get("ambiguity_tags", []))

    "get_pipeline_status": _json_keys(["stage", "complete"]),
    "init_search_state_tool": lambda r, kw: {"run_id": kw.get("run_id")},
    "optimize_routing_prompt": lambda r, kw: {"status": "ok"},

# ---------------------------------------------------------------------------
# Decorator
# ---------------------------------------------------------------------------

def trace_tool(fn: Any) -> Any:
    """Wrap an async MCP tool handler with tracing.

    No-op when ODYSSEUS_TRACE is unset.

    Decorator order (REQUIRED):
        @mcp.tool()   ← outermost: FastMCP sees original signature
        @trace_tool   ← inner: wraps the actual function
        async def my_tool(...):
            ...
    """
    @functools.wraps(fn)
    async def wrapper(*args: Any, **kwargs: Any) -> Any:
        if not os.environ.get("ODYSSEUS_TRACE"):
            return await fn(*args, **kwargs)

        tool_name = fn.__name__
        run_id: str | None = kwargs.get("run_id") or None
        loggable = {k: v for k, v in kwargs.items() if k != "ctx"}

        tracer = _get_tracer()
        seq = tracer.tool_start(tool_name, run_id, loggable)
        t0 = time.monotonic()

        try:
            result = await fn(*args, **kwargs)
            duration_ms = int((time.monotonic() - t0) * 1000)
            extractor = _RESULT_SUMMARIES.get(tool_name, lambda r, kw: {})
            summary = extractor(result if isinstance(result, str) else str(result), kwargs)
            tracer.tool_end(seq, tool_name, run_id, summary, duration_ms)
            return result
        except Exception as exc:
            duration_ms = int((time.monotonic() - t0) * 1000)
            tracer.tool_error(seq, tool_name, run_id, f"{type(exc).__name__}: {exc}", duration_ms)
            raise

    return wrapper
```

```python
import json as _json
import sys as _sys


def _format_ms(ms: int) -> str:
    return f"{ms}ms"


def _format_waterfall(events: list[dict], verbose: bool) -> str:
    """Format paired tool events as a waterfall table."""
    from datetime import datetime

    starts: dict[int, dict] = {}
    ends: dict[int, dict] = {}
    for e in events:
        seq = e["seq"]
        if e["event"] == "tool_start":
            starts[seq] = e
        else:
            ends[seq] = e

    rows = []
    first_ts: datetime | None = None
    total_ms = 0
    error_count = 0

    for seq in sorted(starts):
        start = starts[seq]
        end = ends.get(seq, {})
        tool = start["tool"]
        duration_ms = end.get("duration_ms", 0)
        status = end.get("status", "?")
        error = end.get("error", "")
        args = start.get("args", {})
        summary = end.get("result_summary", {})

        try:
            ts = datetime.fromisoformat(start["timestamp"])
            if first_ts is None:
                first_ts = ts
        except Exception:
            pass

        total_ms += duration_ms
        if status == "error":
            error_count += 1

        rows.append((seq, tool, duration_ms, status, error, args, summary))

    header_ts = first_ts.strftime("%d %b %Y %H:%M:%S") if first_ts else "unknown"
    run_id = events[0].get("run_id", "?") if events else "?"

    lines = [
        f"Run: {run_id}  ·  {header_ts}",
        "",
        f" {'#':>3}  {'Tool':<35} {'Duration':>9}   Status",
        "─" * 60,
    ]

    for seq, tool, duration_ms, status, error, args, summary in rows:
        status_str = "✓" if status == "ok" else f"✗  {error}"
        lines.append(f" {seq:>3}  {tool:<35} {_format_ms(duration_ms):>9}   {status_str}")

        if verbose:
            for k, v in args.items():
                lines.append(f"      args:    {k}: {v}")
            if summary and status == "ok":
                summary_str = "  ".join(f"{k}: {v}" for k, v in summary.items() if v is not None)
                lines.append(f"      result:  {summary_str}")

    lines.append("─" * 60)
    total_s = total_ms / 1000
    summary_line = f"Total: {len(rows)} calls  ·  {total_s:.1f}s"
    if error_count:
        summary_line += f"  ·  {error_count} error{'s' if error_count > 1 else ''}"
    lines.append(summary_line)
    return "\n".join(lines)


def _list_traces(project_dir: Path) -> list[Path]:
    """Return all trace files: run-scoped and session fallback."""
    run_files = sorted(project_dir.glob("outputs/*/traces/events.jsonl"))
    session_files = sorted(project_dir.glob("outputs/traces/session-*.jsonl"))
    return run_files + session_files


def run_trace(run_id: str | None, verbose: bool, list_all: bool) -> None:
    """Execute the trace subcommand."""
    from odysseus.project_dir import get_project_dir

    project_dir = get_project_dir()

    if list_all:
        files = _list_traces(project_dir)
        if not files:
            print("No trace files found.")
            return
        for f in files:
            size_kb = f.stat().st_size // 1024
            parts = list(f.parts)
            if "traces" in parts:
                idx = parts.index("traces")
                label = parts[idx - 1] if parts[idx - 1] != "outputs" else f.stem
            else:
                label = f.stem
            print(f"  {label:<40} {size_kb:>4} KB  {f}")
        return

    if run_id is None:
        print("Error: run_id required (or use --list)", file=_sys.stderr)
        raise SystemExit(1)

    trace_file = project_dir / "outputs" / run_id / "traces" / "events.jsonl"
    if not trace_file.is_file():
        print(f"Error: no trace file found for run '{run_id}' at {trace_file}", file=_sys.stderr)
        raise SystemExit(1)

    events = [_json.loads(line) for line in trace_file.read_text().splitlines() if line.strip()]
    print(_format_waterfall(events, verbose=verbose))
```
